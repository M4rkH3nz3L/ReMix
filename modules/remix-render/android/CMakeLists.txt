# 🔇 V8: AI-zajszűrés (RNNoise v0.2, xiph — BSD-3) — ugyanazok a C-források, mint iOS-en (../rnnoise)
cmake_minimum_required(VERSION 3.22.1)
project(remixrnnoise C)
set(RN_DIR ${CMAKE_CURRENT_SOURCE_DIR}/../rnnoise)
file(GLOB RN_SOURCES CONFIGURE_DEPENDS ${RN_DIR}/*.c)
add_library(remixrnnoise SHARED ${RN_SOURCES} src/main/cpp/rnnoise_jni.c)
target_include_directories(remixrnnoise PRIVATE ${RN_DIR} ${CMAKE_CURRENT_SOURCE_DIR}/../ios)
target_compile_definitions(remixrnnoise PRIVATE USE_WEIGHTS_FILE=1)
target_compile_options(remixrnnoise PRIVATE -O3)
target_link_libraries(remixrnnoise m)
